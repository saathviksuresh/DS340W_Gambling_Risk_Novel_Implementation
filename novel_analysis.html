normalizeToBitcoin <- function(user_bets) {
  user_bets %>% 
    mutate(
      bet_value = bet_value * 1e-8,
      answer_value = answer_value * 1e-8
    )
}

aggregateUser <- function(user_bets) {
  user_bets %>% 
    mutate(
      last_outcome = answer_value - bet_value,
      win_answer_value = ifelse(
        status == "win",
        answer_value - bet_value,
        0
      ),
      loose_answer_value = ifelse(
        status == "loss",
        answer_value - bet_value,
        0
      )
    ) %>%
    mutate(
      dice_balance = cumsum(answer_value - bet_value),
      winsum = cumsum(win_answer_value),
      loosesum = cumsum(loose_answer_value)
    )
}

addlag <- function(user_bets) {
  user_bets %>% 
    mutate(
      bet_pred = lead(bet_value, n = 1),
      game_pred = lead(odds, n = 1),
      time_nextbet = lead(time_bet, n = 1),
      status_pred = lead(status, n = 1)
    )
}

extractAndPrepareUser <- function(x, ...) {
  
  ret <- x %>%
    arrange(time_bet) %>% 
    normalizeToBitcoin() %>% 
    addlag() %>% 
    aggregateUser() %>% 
    left_join(pwin_to_prob, by = "odds") %>%
    left_join(
      pwin_to_prob %>% rename(pwin_pred = pwin),
      by = c("game_pred" = "odds")
    ) %>%
    mutate(
      bet_changed = bet_pred != bet_value,
      game_changed = game_pred != odds,
      exp_changed = bet_pred * game_pred != bet_value * odds,
      
      bet_changed_factor =
        sign(bet_value - bet_pred) %>% factor(levels = c(0, 1, -1)),
      
      game_changed_factor =
        sign(game_pred - odds) %>% factor(levels = c(0, 1, -1)),
      
      exp_changed_factor =
        sign(bet_pred * game_pred - bet_value * odds) %>%
        factor(levels = c(0, 1, -1)),
      
      expectation = bet_pred * game_pred,
      exp_change_ratio =
        log(bet_pred * game_pred / (bet_value * odds)),
      bet_change_ratio =
        log(bet_pred / bet_value),
      game_diff_ratio =
        log(pwin_pred / pwin),
      
      win_cumm_log =
        log10(1 + abs(winsum)),
      loose_cumm_log =
        log10(1 + abs(loosesum)),
      dice_balance_log =
        log10(1 + abs(dice_balance)),
      dice_balance_sign =
        as.character(sign(dice_balance)),
      
      streak = 0,
      difftime = time_nextbet - time_bet,
      stayed =
        !((time_nextbet - time_bet) > 4 * 3600) &
        (timediff < 3600),
      stay_length = 1,
      game_streak = 0,
      streak_total_length = 0,
      streak_balance = 0
    ) %>%
    drop_na()

  for (i in 1:nrow(ret)) {
    
    if (i == 1) {
      
      ret$streak[i] = 0
      ret$stay_length[i] = 1
      ret$game_streak[i] = 0
      ret$streak_balance[i] = 0
      
    } else {
      
      # Win/loss streak
      if (ret$status[i] == ret$status[i - 1]) {
        ret$streak[i] =
          ret$streak[i - 1] + 1
        
        ret$streak_balance[i] =
          ret$streak_balance[i - 1] +
          ret$answer_value[i] -
          ret$bet_value[i]
        
      } else {
        ret$streak[i] = 0
        ret$streak_balance[i] = 0
      }
      
      # Session/stay length
      if (ret$stayed[i]) {
        ret$stay_length[i] =
          ret$stay_length[i - 1] + 1
      } else {
        ret$stay_length[i] = 1
      }
      
      # Same-game streak
      if (!ret$game_changed[i]) {
        ret$game_streak[i] =
          ret$game_streak[i - 1] + 1
      } else {
        ret$game_streak[i] = 0
      }
    }
  }

  for (i in nrow(ret):1) {
    
    if (i == nrow(ret)) {
      
      ret$streak_total_length[i] =
        ret$streak[i]
      
      streak_total_length =
        ret$streak[i]
      
    } else {
      
      if (streak_total_length & ret$streak[i]) {
        
        ret$streak_total_length[i] =
          streak_total_length
        
      } else if (streak_total_length & !ret$streak[i]) {
        
        ret$streak_total_length[i] =
          streak_total_length
        
        streak_total_length = 0
        
      } else if (!streak_total_length & ret$streak[i]) {
        
        streak_total_length =
          ret$streak[i]
        
        ret$streak_total_length[i] =
          streak_total_length
        
      } else {
        
        ret$streak_total_length[i] =
          streak_total_length
      }
    }
  }

  return(
    ret %>%
      arrange(time_bet)
  )
}

prepare_user_table <- function(user_bets_matched, min_games = 2, ...) {
  
  user_bets_matched %>% 
    drop_na() %>%
    group_by(userID) %>% 
    mutate(ngamez = n()) %>% 
    filter(ngamez > min_games) %>%
    group_modify(extractAndPrepareUser) %>%
    ungroup()
}

create_analysis_sets <- function(prepared_sample, active_day_map, eligible_players) {

  analysis_players <- active_day_map %>%
    semi_join(eligible_players, by = "userID") %>%
    group_by(userID) %>%
    filter(max(active_day) >= 4) %>%
    ungroup()

  transactions_with_day <- prepared_sample %>%
    mutate(
      date = as.POSIXct(time_bet, origin = "1970-01-01") %>%
        as.Date()
    ) %>%
    inner_join(
      analysis_players %>%
        select(userID, date, active_day),
      by = c("userID", "date")
    )

  observation_set <- transactions_with_day %>%
    filter(active_day %in% 1:3)

  target_set <- transactions_with_day %>%
    filter(active_day %in% 4:10)

  list(
    observation = observation_set,
    target = target_set
  )
}

# Nagelkerke R-squared
nagelkerke_r2 <- function(model) {

  n <- nobs(model)

  ll_model <- as.numeric(logLik(model))

  null_model <- update(
    model,
    . ~ 1
  )

  ll_null <- as.numeric(logLik(null_model))

  cox_snell <- 1 - exp(
    (2 / n) * (ll_null - ll_model)
  )

  nagelkerke <- cox_snell /
    (1 - exp((2 / n) * ll_null))

  nagelkerke
}

shap_type_importance <- shap_importance %>%
  group_by(feature_type) %>%
  summarise(
    total_mean_abs_shap = sum(mean_abs_shap),
    .groups = "drop"
  ) %>%
  mutate(
    proportion = total_mean_abs_shap / sum(total_mean_abs_shap)
  ) %>%
  mutate(
    across(
      where(is.numeric),
      ~ round(.x, 3)
    )
  )
